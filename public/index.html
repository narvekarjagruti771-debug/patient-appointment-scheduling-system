1. SQL Database View

CREATE VIEW upcoming_appointments_view AS
SELECT 
    a.appointment_id,
    a.patient_id,
    p.name AS patient_name,
    p.phone AS patient_phone,
    a.doctor_id,
    d.name AS doctor_name,
    d.specialization AS doctor_specialization,
    a.appointment_date,
    a.appointment_time,
    a.status,
    a.reason
FROM appointments a
JOIN patients p ON a.patient_id = p.patient_id
JOIN doctors d ON a.doctor_id = d.doctor_id
ORDER BY a.appointment_date ASC, a.appointment_time ASC;

2. SQL Database Triggers

-- Log on appointment creation
CREATE TRIGGER after_appointment_insert
AFTER INSERT ON appointments
FOR EACH ROW
BEGIN
    INSERT INTO appointment_logs (appointment_id, action_type, old_status, new_status, description)
    VALUES (NEW.appointment_id, 'CREATE', NULL, NEW.status, 
    CONCAT('New appointment booked for patient ID ', NEW.patient_id, ' with doctor ID ', NEW.doctor_id, ' for ', NEW.appointment_date));
END;

-- Log on status changes
CREATE TRIGGER after_appointment_update
AFTER UPDATE ON appointments
FOR EACH ROW
BEGIN
    IF OLD.status <> NEW.status THEN
        INSERT INTO appointment_logs (appointment_id, action_type, old_status, new_status, description)
        VALUES (NEW.appointment_id, 'UPDATE', OLD.status, NEW.status, 
        CONCAT('Appointment status changed from ', OLD.status, ' to ', NEW.status));
    END IF;
END;

3. SQL Stored Procedure with Cursor

CREATE PROCEDURE generate_daily_report(IN target_date DATE)
BEGIN
    DECLARE done INT DEFAULT FALSE;
    DECLARE appt_time TIME;
    DECLARE p_name VARCHAR(100);
    DECLARE d_name VARCHAR(100);
    DECLARE report_text TEXT DEFAULT '';
    
    -- Cursor to loop appointments for date
    DECLARE appt_cursor CURSOR FOR
        SELECT a.appointment_time, p.name, d.name 
        FROM appointments a
        JOIN patients p ON a.patient_id = p.patient_id
        JOIN doctors d ON a.doctor_id = d.doctor_id
        WHERE a.appointment_date = target_date;
        
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;
    
    OPEN appt_cursor;
    read_loop: LOOP
        FETCH appt_cursor INTO appt_time, p_name, d_name;
        IF done THEN LEAVE read_loop; END IF;
        
        SET report_text = CONCAT(report_text, 'Appt at ', appt_time, ': ', p_name, ' with Dr. ', d_name, '\n');
    END LOOP;
    CLOSE appt_cursor;
    
    -- Save compiled report
    INSERT INTO daily_reports (report_date, report_summary)
    VALUES (target_date, report_text)
    ON DUPLICATE KEY UPDATE report_summary = VALUES(report_summary);
END;
